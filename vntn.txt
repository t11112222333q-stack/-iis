local function GetBoss()
    local b = workspace:FindFirstChild("Boss Zone") and workspace["Boss Zone"]:FindFirstChild("Boss")
    return (b and b:FindFirstChild("Hitbox") and b:FindFirstChildOfClass("Humanoid").Health > 0) and b or nil
end

local function ScanForPlayers()
    local rootPart = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return false, nil end
    for _, otherPlayer in pairs(Players:GetPlayers()) do
        if otherPlayer ~= localPlayer and otherPlayer.Character and otherPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local distance = (rootPart.Position - otherPlayer.Character.HumanoidRootPart.Position).Magnitude
            if distance < BAN_KINH_SCAN_NGUOI then return true, otherPlayer.Name end
        end
    end
    return false, nil
end

            if s.playing < s.maxPlayers and s.id ~= game.JobId then 
                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, localPlayer) 
                break 
            end 
        end
    end)
end

local function FormatNumber(num)
    local formatted = tostring(num)
    while true do  
        formatted, k = string.gsub(formatted, "^(-?%d+)(%d%d%d)", '%1,%2')
        if k == 0 then break end
    end
    return formatted
end

local function deepAnalysis()
    if #history < 3 then return "Ga", 50 end
    local scoreGa, scoreGau = 0, 0
    local lastSide = history[#history]

    local l1, l2, l3 = history[#history], history[#history-1], history[#history-2]
    if l3 == "Ga" and l2 == "Gau" and l1 == "Ga" then
        scoreGau = scoreGau + 4.5
        aiMsgParagraph:SetDesc("Em guộc AI: Phát hiện chuẩn chu kỳ GÀ - GẤU - GÀ!")
    end

    local streak = 1
    for i = #history - 1, 1, -1 do
        if history[i] == lastSide then streak = streak + 1 else break end
    end
    if streak >= 3 then
        local breakMultiplier = 1.0 + (streak * 1.5)
        if lastSide == "Ga" then
            scoreGau = scoreGau + breakMultiplier
            aiMsgParagraph:SetDesc(string.format("Em guộc AI: Gà bệt %d cây rồi, sòng chuẩn bị hồi Gấu, lực bẻ đang tăng!", streak))
        else
            scoreGa = scoreGa + breakMultiplier
            aiMsgParagraph:SetDesc(string.format("Em guộc AI: Gấu bệt dai quá (%d cây), em tính toán lực ép bẻ Gà cực căng!", streak))
        end
    end

    local cGa, cGau = 0, 0
    local scanRange = math.min(25, #history)
    for i = #history - scanRange + 1, #history do
        if history[i] == "Ga" then cGa = cGa + 1 else cGau = cGau + 1 end
    end
    if cGa ~= cGau then
        local densityBonus = math.abs(cGa - cGau) * 0.7
        if cGa > cGau then scoreGau = scoreGau + densityBonus else scoreGa = scoreGa + densityBonus end
    end

    local m1, m2 = history[#history], history[#history-1]
    local tGa, tGau = 0, 0
    for i = 1, #history - 2 do
        if history[i] == m2 and history[i+1] == m1 then
            if history[i+2] == "Ga" then tGa = tGa + 1 else tGau = tGau + 1 end
        end
    end
    if tGa + tGau > 0 then
        if tGa > tGau then scoreGau = scoreGau + 1.2 else scoreGa = scoreGa + 1.2 end
    end

    local totalScore = scoreGa + scoreGau
    if totalScore == 0 then return (lastSide == "Ga" and "Gau" or "Ga"), 50 end
    return (scoreGa > scoreGau) and "Ga" or "Gau", 50 + math.floor((math.max(scoreGa, scoreGau) / totalScore) * 45)
end

            if lastPred ~= "" and betPlaced then

                end
            end
            table.insert(history, clean) if #history > 35 then table.remove(history, 1) end
            updateGaGauUI() lastPred = "" betPlaced = false
        elseif act == "SyncState" then

local function getClosestPlayer()

    local localChar = localPlayer.Character local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")

    for _, p in pairs(Players:GetPlayers()) do

            if p.Character.Humanoid.Health > 0 then

local function HopServerCustom(targetMode)
    Fluent:Notify({ Title = "Server Hop", Content = "Đang quét danh sách máy chủ khả thi...", Duration = 3 })
    pcall(function()
        local placeId = game.PlaceId
        local currentJobId = game.JobId
        local cursor = ""
        local candidateServer = nil
        local bestCount = (targetMode == "Fullest") and -1 or 999
        
        repeat
            local url = "https://games.roblox.com/v1/games/" .. placeId .. "/servers/0?sortOrder=Desc&limit=100" .. (cursor ~= "" and "&cursor=" .. cursor or "")
            local req = request or http_request or (syn and syn.request) or (http and http.request)
            if not req then break end
            
            local res = req({ Url = url, Method = "GET" })
            local body = HttpService:JSONDecode(res.Body)
            
            if body and body.data then
                for _, s in ipairs(body.data) do
                    if s.id ~= currentJobId and s.playing < s.maxPlayers and s.playing > 0 then
                        if targetMode == "Fullest" then
                            if s.playing > bestCount then
                                bestCount = s.playing
                                candidateServer = s.id
                            end
                        elseif targetMode == "Lowest" then
                            if s.playing < bestCount then
                                bestCount = s.playing
                                candidateServer = s.id
                            end
                        end
                    end
                end
            end
            cursor = body and body.nextPageCursor or nil
        until not cursor or candidateServer ~= nil
        
        if candidateServer then
            TeleportService:TeleportToPlaceInstance(placeId, candidateServer, localPlayer)
        else
            Fluent:Notify({ Title = "Server Hop", Content = "Không tìm thấy máy chủ thích hợp!", Duration = 3 })
        end
    end)
end

-- HEARTBEAT MAIN LOOP & CORE LOGIC COMBAT ESP
-- ============================================================================
RunService.Heartbeat:Connect(function()
    local char = localPlayer.Character local root = char and char:FindFirstChild("HumanoidRootPart") local hum = char and char:FindFirstChild("Humanoid")
    if not char or not root or not hum then return end
    
    if _G.SpeedPvP then
        if hum.MoveDirection.Magnitude > 0 then
            local targetVelocity = hum.MoveDirection * getgenv().BypassVelocitySpeed
            root.AssemblyLinearVelocity = Vector3.new(targetVelocity.X, root.AssemblyLinearVelocity.Y, targetVelocity.Z)
        else
            root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)
        end
    end
    
    if _G.AimlockPvP and not aimPlayerState then
        local target = getClosestPlayer()
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            local targetPos = target.Character.HumanoidRootPart.Position
            root.CFrame = CFrame.new(root.Position, Vector3.new(targetPos.X, root.Position.Y, targetPos.Z))
        end
    end
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local targetRoot = p.Character:FindFirstChild("HumanoidRootPart")
            if targetRoot then
                if _G.HitboxPvP then
                    targetRoot.Size = HITBOX_SIZE_RED targetRoot.Transparency = 0.75 targetRoot.Color = Color3.fromRGB(244, 63, 94) targetRoot.CanCollide = true
                else
                    if targetRoot.Size == HITBOX_SIZE_RED then targetRoot.Size = Vector3.new(2, 2, 1) targetRoot.Transparency = 1 targetRoot.CanCollide = true end
                end
            end
            if toolHitboxESPState then
                for _, child in ipairs(p.Character:GetChildren()) do
                    if child:IsA("Tool") then
                        local handle = child:FindFirstChild("Handle") or child:FindFirstChild("Hitbox")
                        if handle and not toolBoxes[child] then
                            local sBox = Instance.new("SelectionBox", handle)
                            sBox.Name = "GlobalToolHitbox" sBox.Adornee = handle sBox.LineThickness = 0.04 sBox.Color3 = Color3.fromRGB(0, 255, 0) sBox.SurfaceTransparency = 0.9
                            toolBoxes[child] = sBox
                        end
                    end
                end
            end
            if healthESPState then
                applyHealthESP(p.Character)
                local tHum = p.Character:FindFirstChildOfClass("Humanoid") local tHead = p.Character:FindFirstChild("Head")
                if tHum and tHead and tHead:FindFirstChild("HealthESP") then
                    local esp = tHead.HealthESP local bar = esp.bgFrame.HealthBar local percentage = math.clamp(tHum.Health / tHum.MaxHealth, 0, 1)
                    bar.Size = UDim2.new(percentage, 0, 1, 0)
                    bar.BackgroundColor3 = percentage > 0.6 and Color3.fromRGB(0, 255, 0) or (percentage > 0.3 and Color3.fromRGB(255, 255, 0) or Color3.fromRGB(255, 0, 0))
                    esp.HPText.Text = math.floor(tHum.Health) .. "/" .. math.floor(tHum.MaxHealth)
                end
            end
        end
    end
end)

Players.PlayerRemoving:Connect(function(p)
    if p.Character then for _, child in ipairs(p.Character:GetChildren()) do if toolBoxes[child] then toolBoxes[child]:Destroy() toolBoxes[child] = nil end end end
end)
