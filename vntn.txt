local function getClosestPlayer()
    local closestPlayer = nil local shortestDistance = AIMLOCK_RADIUS
    local localChar = localPlayer.Character local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")
    if not localRoot then return nil end
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") then
            if p.Character.Humanoid.Health > 0 then
                local distance = (localRoot.Position - p.Character.HumanoidRootPart.Position).Magnitude
                if distance < shortestDistance then shortestDistance = distance closestPlayer = p end
            end
        end
    end
    return closestPlayer
end

local function ScanForPlayers()
    local rootPart = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return false, nil end
    for _, otherPlayer in pairs(Players:GetPlayers()) do
        if otherPlayer ~= localPlayer and otherPlayer.Character and otherPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local distance = (rootPart.Position - otherPlayer.Character.HumanoidRootPart.Position).Magnitude
            if distance < BAN_KINH_SCAN_NGUOI then return true, otherPlayer.Name end
        end
    end
    return false, nil
end

local function HopToLowestServer()
    Fluent:Notify({ Title = "Hệ thống", Content = "Đang quét và tự động chuyển đổi Server an toàn...", Duration = 3 })
    pcall(function()
        local servers = HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/"..game.PlaceId.."/servers/Public?limit=100")).data
        for _, s in pairs(servers) do 
            if s.playing < s.maxPlayers and s.id ~= game.JobId then 
                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, localPlayer) 
                break 
            end 
        end
    end)
end

local function execBet(side, amt)
    if not autoGaGau then return end
    task.spawn(function()
        pcall(function()
            local gaGauGui = localPlayer.PlayerGui:FindFirstChild("GaGauGui")
            if not gaGauGui then return end
            local frameGaGau = gaGauGui:FindFirstChild("FrameGaGau")
            if not frameGaGau then return end

            local targetBtn = (side == "Ga") and "ChonGa" or "ChonGau"
            local targetInput = (side == "Ga") and "DienSoTienGa" or "DienSoTienGau"
            local chon = frameGaGau:FindFirstChild(targetBtn)
            local dienSoTien = frameGaGau:FindFirstChild(targetInput)
            local xacNhan = frameGaGau:FindFirstChild("XacNhan")

            if chon and dienSoTien and xacNhan then
                for _, conn in pairs(getconnections(chon.MouseButton1Click)) do conn:Fire() end
                task.wait(0.08)
                dienSoTien.Text = tostring(math.floor(amt))
                task.wait(0.08)
                for _, conn in pairs(getconnections(xacNhan.MouseButton1Click)) do conn:Fire() end
            end
        end)
    end)
end

local function deepAnalysis()
    if #history < 3 then return "Ga", 50 end
    
    local scoreGa, scoreGau = 0, 0
    local lastSide = history[#history]

    -- TẦNG 1: CHU KỲ CHUẨN BÀI 1-1 (Gà -> Gấu -> Gà)
    local l1, l2, l3 = history[#history], history[#history-1], history[#history-2]
    -- Thứ tự thời gian từ cũ đến mới: l3 (Gà) -> l2 (Gấu) -> l1 (Gà)
    if l3 == "Ga" and l2 == "Gau" and l1 == "Ga" then
        scoreGau = scoreGau + 4.5
        aiMsgParagraph:SetDesc("🤖 Em guộc AI: Phát hiện chuẩn chu kỳ GÀ - GẤU - GÀ! Theo nhịp đi thằng lồn!")
    end

    -- TẦNG 2: BẺ BỆT DÀI
    local streak = 1
    for i = #history - 1, 1, -1 do
        if history[i] == lastSide then streak = streak + 1 else break end
    end
    if streak >= 3 then
        local breakMultiplier = 1.0 + (streak * 1.5)
        if lastSide == "Ga" then
            scoreGau = scoreGau + breakMultiplier
            aiMsgParagraph:SetDesc(string.format("🤖 Em guộc AI: Gà bệt %d cây rồi, sòng chuẩn bị hồi Gấu, lực bẻ đang tăng!", streak))
        else
            scoreGa = scoreGa + breakMultiplier
            aiMsgParagraph:SetDesc(string.format("🤖 Em guộc AI: Gấu bệt dai quá (%d cây), em tính toán lực ép bẻ Gà cực căng!", streak))
        end
    end

    -- TẦNG 3: MẬT ĐỘ DIỆN RỘNG
    local cGa, cGau = 0, 0
    local scanRange = math.min(25, #history)
    for i = #history - scanRange + 1, #history do
        if history[i] == "Ga" then cGa = cGa + 1 else cGau = cGau + 1 end
    end
    if cGa ~= cGau then
        local densityBonus = math.abs(cGa - cGau) * 0.7
        if cGa > cGau then scoreGau = scoreGau + densityBonus else scoreGa = scoreGa + densityBonus end
    end

    -- TẦNG 4 & 5: DỮ LIỆU VẾT KHỚP CHUỖI KHÉP KÍN
    local m1, m2 = history[#history], history[#history-1]
    local tGa, tGau = 0, 0
    for i = 1, #history - 2 do
        if history[i] == m2 and history[i+1] == m1 then
            if history[i+2] == "Ga" then tGa = tGa + 1 else tGau = tGau + 1 end
        end
    end
    if tGa + tGau > 0 then
        if tGa > tGau then scoreGau = scoreGau + 1.2 else scoreGa = scoreGa + 1.2 end
    end

    local totalScore = scoreGa + scoreGau
    if totalScore == 0 then return (lastSide == "Ga" and "Gau" or "Ga"), 50 end
    
    if scoreGa > scoreGau then
        return "Ga", 50 + math.floor((scoreGa / totalScore) * 45)
    else
        return "Gau", 50 + math.floor((scoreGau / totalScore) * 45)
    end
end

local function updateGaGauUI()
    local ga, gau, emo = 0, 0, {}
    for i = math.max(1, #history - 9), #history do
        if history[i] == "Ga" then ga = ga + 1; table.insert(emo, "🐥") else gau = gau + 1; table.insert(emo, "🐻") end
    end
    
    scoreTracker:SetDesc(string.format("🐥 Gà: %d | 🐻 Gấu: %d\n📜 Lịch sử mới: %s", ga, gau, #emo > 0 and table.concat(emo, " ") or "[---]"))
    
    local leader = localPlayer:FindFirstChild("leaderstats")
    local vnd = leader and leader:FindFirstChild("VND")
    local balance = vnd and vnd.Value or 0
    
    pulseTracker:SetDesc(string.format("💰 Số Dư Tài Khoản: %s VND\n🏆 Thắng: %d/%d | Thua liên tiếp: %d\n💵 Cược Hiện Tại: %s", 
        string.format("%.0f", balance), wins, winTarget, loses, (betCur >= 1e6 and string.format("%.1fM", betCur / 1e6) or tostring(betCur))))
end

-- REALTIME NETWORK SYNC GA GAU
task.spawn(function()
    local remote = ReplicatedStorage:WaitForChild("Remote", 10)
    if not remote then return end
    local upd = remote:WaitForChild("GaGauUpdate", 10)
    if not upd then return end

    upd.OnClientEvent:Connect(function(act, ...)
        local args = { ... }
        if act == "Countdown" then
            local sec = args[1]
            updateGaGauUI()
            
            if sec == 25 and autoGaGau and not betPlaced and not frozen then
                local side, conf = deepAnalysis()
                if conf >= confThres then
                    lastPred = side
                    betPlaced = true
                    predictionParagraph:SetDesc(string.format("🔮 CHỌN: %s | 📊 ĐỘ TỰ TIN: %d%%", side == "Ga" and "🐥 GÀ" or "🐻 GẤU", conf))
                    execBet(side, betCur)
                else
                    predictionParagraph:SetDesc(string.format("🔮 BỎ QUA PHIÊN NÀY (%d%% < %d%%)", conf, confThres))
                end
            end
        elseif act == "Result" then
            local winner = args[1]
            if not winner then return end
            local clean = (string.lower(winner) == "gau") and "Gau" or "Ga"

            if lastPred ~= "" and betPlaced then
                if lastPred == clean then
                    wins = wins + 1
                    loses = 0
                    betCur = betBase
                    frozen = false
                    aiMsgParagraph:SetDesc("🤖 Em guộc AI: Địt mẹ game tuổi con cặc ngon ko em.")
                else
                    loses = loses + 1
                    if loses >= loseMax then
                        frozen = true
                        betCur = betBase
                        aiMsgParagraph:SetDesc("🤖 Em guộc AI: Cái lồn mẹ nó game như cặc! Chạm giới hạn lỗ!")
                    else
                        betCur = betCur * betMulti
                        aiMsgParagraph:SetDesc("🤖 Em guộc AI: Cầu gãy rồi thằng ngu để bố tính!")
                    end
                end
            end

            table.insert(history, clean)
            if #history > 35 then table.remove(history, 1) end
            updateGaGauUI()

            lastPred = ""
            betPlaced = false
            
            if wins >= winTarget then
                autoGaGau = false
                toggleGG:SetValue(false)
                aiMsgParagraph:SetDesc("🤖 Em guộc AI: Nghỉ đi thằng lồn địt mẹ tham thì thua! Đủ chỉ tiêu!")
                predictionParagraph:SetDesc("🏆 HOÀN THÀNH CHỈ TIÊU!")
            end
        elseif act == "SyncState" then
            local data = args[1] or {}
            if data.history then
                table.clear(history)
                for _, v in ipairs(data.history) do
                    table.insert(history, (string.lower(tostring(v)) == "gau") and "Gau" or "Ga")
                end
                while #history > 35 do table.remove(history, 1) end
                updateGaGauUI()
            end
        end
    end)
    upd:FireServer("RequestSync")
end)

RunService.Heartbeat:Connect(function()
    local char = localPlayer.Character local root = char and char:FindFirstChild("HumanoidRootPart") local hum = char and char:FindFirstChild("Humanoid")
    if not char or not root or not hum then return end
    
    if _G.SpeedPvP then
        if hum.MoveDirection.Magnitude > 0 then
            local targetVelocity = hum.MoveDirection * getgenv().BypassVelocitySpeed
            root.AssemblyLinearVelocity = Vector3.new(targetVelocity.X, root.AssemblyLinearVelocity.Y, targetVelocity.Z)
        else
            root.AssemblyLinearVelocity = Vector3.new(0, root.AssemblyLinearVelocity.Y, 0)
        end
    end
    
    if _G.AimlockPvP and not aimPlayerState then
        local target = getClosestPlayer()
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            local targetPos = target.Character.HumanoidRootPart.Position
            root.CFrame = CFrame.new(root.Position, Vector3.new(targetPos.X, root.Position.Y, targetPos.Z))
        end
    end
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character then
            local targetRoot = p.Character:FindFirstChild("HumanoidRootPart")
            if targetRoot then
                if _G.HitboxPvP then
                    targetRoot.Size = HITBOX_SIZE_RED targetRoot.Transparency = 0.75 targetRoot.Color = Color3.fromRGB(244, 63, 94) targetRoot.CanCollide = true
                else
                    if targetRoot.Size == HITBOX_SIZE_RED then targetRoot.Size = Vector3.new(2, 2, 1) targetRoot.Transparency = 1 targetRoot.CanCollide = true end
                end
            end
            if toolHitboxESPState then
                for _, child in ipairs(p.Character:GetChildren()) do
                    if child:IsA("Tool") then
                        local handle = child:FindFirstChild("Handle") or child:FindFirstChild("Hitbox")
                        if handle and not toolBoxes[child] then
                            local sBox = Instance.new("SelectionBox", handle)
                            sBox.Name = "GlobalToolHitbox" sBox.Adornee = handle sBox.LineThickness = 0.04 sBox.Color3 = Color3.fromRGB(0, 255, 0) sBox.SurfaceTransparency = 0.9
                            toolBoxes[child] = sBox
                        end
                    end
                end
            end
            if healthESPState then
                applyHealthESP(p.Character)
                local tHum = p.Character:FindFirstChildOfClass("Humanoid") local tHead = p.Character:FindFirstChild("Head")
                if tHum and tHead and tHead:FindFirstChild("HealthESP") then
                    local esp = tHead.HealthESP local bar = esp.bgFrame.HealthBar local percentage = math.clamp(tHum.Health / tHum.MaxHealth, 0, 1)
                    bar.Size = UDim2.new(percentage, 0, 1, 0)
                    bar.BackgroundColor3 = percentage > 0.6 and Color3.fromRGB(0, 255, 0) or (percentage > 0.3 and Color3.fromRGB(255, 255, 0) or Color3.fromRGB(255, 0, 0))
                    esp.HPText.Text = math.floor(tHum.Health) .. "/" .. math.floor(tHum.MaxHealth)
                end
            end
        end
    end
end)

Players.PlayerRemoving:Connect(function(p)
    if p.Character then for _, child in ipairs(p.Character:GetChildren()) do if toolBoxes[child] then toolBoxes[child]:Destroy() toolBoxes[child] = nil end end end
